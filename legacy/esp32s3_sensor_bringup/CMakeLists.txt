cmake_minimum_required(VERSION 3.22)
if(NOT MTKERNEL_PATH)
    set(MTKERNEL_PATH "$ENV{MTKERNEL_PATH}")
endif()
if(NOT EXISTS "${MTKERNEL_PATH}/components/mtkernel/mtk_startup.c")
    message(FATAL_ERROR "Set MTKERNEL_PATH to the real ESP32-S3 SMP kernel port; no FreeRTOS fallback is provided.")
endif()
set(EXTRA_COMPONENT_DIRS "${MTKERNEL_PATH}/components")
set(COMPONENTS main freertos_shim)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(aai15_esp32s3_mtk_sensors)
if(NOT CONFIG_MTK_SMP OR CONFIG_FREERTOS_UNICORE)
    message(FATAL_ERROR "This application requires the dual-core micro T-Kernel profile.")
endif()
